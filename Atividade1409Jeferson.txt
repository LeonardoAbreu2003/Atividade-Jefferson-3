import java.util.Scanner;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int S = sc.nextInt();
        int A = sc.nextInt();
        int B = sc.nextInt();

        int quantidade = 0;

        for (int i = A; i <= B; i++) {
        int numero = i;
        int soma = 0;

        while (numero > 0) {
        soma = soma + (numero % 10);
        numero = numero / 10;
}

        if (soma == S) {
        quantidade++;
    }
}

    System.out.println("A quantidade de números com soma dos dígitos igual a " + S + " é: " + quantidade);
    }
}
